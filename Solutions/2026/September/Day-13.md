## Problem #89
Fibonacci: Calculate the nth Fibonacci number

## Solution
```python
class Solution:
    def fibonacci(self, n):
        """
        Calculate the nth Fibonacci number using Dynamic Programming
        
        Approach: Bottom-up DP with space optimization
        Time Complexity: O(n)
        Space Complexity: O(1)
        """
        if n <= 1:
            return n
        
        # Use two variables to store previous two values
        prev, curr = 0, 1
        
        for _ in range(2, n + 1):
            # Calculate next Fibonacci number
            next_fib = prev + curr
            prev = curr
            curr = next_fib
        
        return curr

# Alternative: Recursive with Memoization
class SolutionMemo:
    def __init__(self):
        self.memo = {}
    
    def fibonacci(self, n):
        """
        Calculate nth Fibonacci using memoization
        Time Complexity: O(n)
        Space Complexity: O(n)
        """
        if n in self.memo:
            return self.memo[n]
        
        if n <= 1:
            return n
        
        self.memo[n] = self.fibonacci(n - 1) + self.fibonacci(n - 2)
        return self.memo[n]

# Alternative: Using List DP
class SolutionListDP:
    def fibonacci(self, n):
        """
        Calculate nth Fibonacci using list for all values
        Time Complexity: O(n)
        Space Complexity: O(n)
        """
        if n <= 1:
            return n
        
        dp = [0] * (n + 1)
        dp[1] = 1
        
        for i in range(2, n + 1):
            dp[i] = dp[i - 1] + dp[i - 2]
        
        return dp[n]
```
