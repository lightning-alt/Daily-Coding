## Problem #73
Palindrome Check: Check if a string is a palindrome

## Solution
```python
class Solution:
    def isPalindrome(self, s):
        """
        Check if a string is a palindrome (ignoring spaces and special chars)
        
        Approach: Two-pointer from both ends, ignore non-alphanumeric
        Time Complexity: O(n)
        Space Complexity: O(1)
        """
        left, right = 0, len(s) - 1
        
        while left < right:
            # Skip non-alphanumeric characters from left
            while left < right and not s[left].isalnum():
                left += 1
            
            # Skip non-alphanumeric characters from right
            while left < right and not s[right].isalnum():
                right -= 1
            
            # Compare characters (case-insensitive)
            if s[left].lower() != s[right].lower():
                return False
            
            left += 1
            right -= 1
        
        return True

# Alternative: Using string filtering and reversing
class SolutionSimple:
    def isPalindrome(self, s):
        """
        Simple palindrome check using filtering
        Time Complexity: O(n)
        Space Complexity: O(n)
        """
        # Filter and convert to lowercase
        filtered = ''.join(c.lower() for c in s if c.isalnum())
        
        # Check if equal to its reverse
        return filtered == filtered[::-1]
```
